\heading{A. KIẾN THỨC TỔNG QUAN VỀ DOANH NGHIỆP}

\tieumuc{Tổng quan về công ty}
\todo{Lĩnh vực hoạt động, trụ sở, quy mô người dùng/thị trường, đặc thù hệ thống dữ liệu.}

\tieumuc{Quá trình hình thành và phát triển}
\todo{Năm thành lập, nhà sáng lập, các mốc mở rộng thị trường và gọi vốn.}

\tieumuc{Mô hình kinh doanh và đặc thù dữ liệu}
\todo{Mô hình kinh doanh; yêu cầu dữ liệu phát sinh từ mô hình đó; bối cảnh dẫn tới bài toán.}

\tieumuc{Tầm nhìn và mục tiêu}
\todo{Tầm nhìn công ty; vai trò của bộ phận nơi sinh viên kiến tập.}
